### 3. Revenue Split (自动分润) - 核心合约逻辑

```solidity
// contracts/RevenueSplitter.sol
contract RevenueSplitter {
    struct SplitConfig {
        address recipient;
        uint256 bps;        // 基点 (10000 bps = 100%)
        bool isActive;
    }
    
    // 分润配置 (可配置多重分润)
    mapping(bytes32 => SplitConfig[]) public splits;
    
    // 四权分润比例预设
    mapping(bytes32 => mapping(uint8 => SplitConfig[])) public rightsSplits;
    // bytes32 = assetId
    // uint8 = 0:所有权, 1:使用权, 2:衍生权, 3:扩展权
    
    event RevenueDistributed(
        bytes32 indexed assetId,
        uint256 totalAmount,
        uint8 rightType,
        address[] recipients,
        uint256[] amounts
    );
    
    function distributeRevenue(
        bytes32 assetId,
        uint256 amount,
        uint8 rightType
    ) external {
        SplitConfig[] storage configs = rightsSplits[assetId][rightType];
        
        address[] memory recipients = new address[](configs.length);
        uint256[] memory amounts = new uint256[](configs.length);
        
        for (uint i = 0; i < configs.length; i++) {
            amounts[i] = (amount * configs[i].bps) / 10000;
            recipients[i] = configs[i].recipient;
            payable(recipients[i]).transfer(amounts[i]);
        }
        
        emit RevenueDistributed(assetId, amount, rightType, recipients, amounts);
    }
}
```

```tsx
// src/components/RightsDashboard/RightsChart.tsx
import React, { useEffect, useState } from 'react'
import { useECHOContract } from '@/hooks/useECHOContract'

interface RightsChartProps {
  assetId: string
}

export const RightsChart: React.FC<RightsChartProps> = ({ assetId }) => {
  const { getRightsDistribution } = useECHOContract()
  const [rights, setRights] = useState<RightsDistribution | null>(null)

  useEffect(() => {
    getRightsDistribution(assetId).then(setRights)
  }, [assetId])

  if (!rights) return <LoadingSpinner />

  return (
    <div className="rights-chart">
      <div className="rights-bars">
        {rights.map((r) => (
          <div key={r.type} className="right-bar">
            <div className="right-label">{r.typeName}</div>
            <div className="bar-container">
              <div 
                className="bar-fill" 
                style={{ width: `${r.percentage}%` }}
              />
            </div>
            <div className="right-value">{r.percentage}%</div>
          </div>
        ))}
      </div>
    </div>
  )
}
```

```typescript
- 验证和处理权能代币
- 执行自动分润
- 管理和调用技能资产

## 使用方法

### 验证使用权
```
@ECHO verify-usage --asset-id <assetId> --amount <amount>
```

### 执行分润
```
@ECHO split-revenue --asset-id <assetId> --amount <amount> --right-type <type>
```

### 铸造技能资产
```
@ECHO mint-skill --name <name> --metadata <metadata> --rights <config>
```

## 配置参数
- `contractAddress`: ECHO合约地址
- `chainId`: 链ID
- `autoConfirm`: 自动确认交易 (默认false)
```
